A **Single-Agent System** relies on one continuous Claude execution loop to reason, call tools, and manage conversation state, whereas a **Multi-Agent Architecture** uses an orchestrator agent to delegate specialized tasks to independent subagents running in isolated context windows.

**1. Single-Agent System (Autonomous Tool Loop)**

In a single-agent system, Claude operates as a generalist in a single context history. It repeatedly evaluates user input, executes requested tools, appends tool results, and continues until it produces a final response.

```python
import anthropic

client = anthropic.Anthropic()
model = "claude-sonnet-4-6"
def execute_tool(name: str, args: dict) -> str:
    if name == "calculator":
        return str(eval(args["expression"]))
    return "Unknown tool"

def run_single_agent(user_query: str):
    tools = [
        {
            "name": "calculator",
            "description": "Perform basic arithmetic.",
            "input_schema": {
                "type": "object",
                "properties": {"expression": {"type": "string"}},
                "required": ["expression"]
            }
        }
    ]
    
    messages = [{"role": "user", "content": user_query}]
    
    while True:
        response = client.messages.create(
            model=model,
            max_tokens=1024,
            tools=tools,
            messages=messages
        )
        
        messages.append({"role": "assistant", "content": response.content})
        
        if response.stop_reason == "end_turn":
            print(f"Final Output: {response.content[0].text}")
            break
            
        if response.stop_reason == "tool_use":
            tool_results = []
            for block in response.content:
                if block.type == "tool_use":
                    result = execute_tool(block.name, block.input)
                    tool_results.append({
                        "type": "tool_result",
                        "tool_use_id": block.id,
                        "content": result
                    })
            
            messages.append({"role": "user", "content": tool_results})

run_single_agent("What is (45 * 12) + 180?")


```

**2. Multi-Agent Architecture (Orchestrator-Worker Pattern)**

In a multi-agent system, an **Orchestrator** decomposes a complex task and delegates sub-problems to specialized worker subagents. Worker logs stay contained inside their own ephemeral context windows, returning only distilled summaries to the main orchestrator.

```python
import anthropic

client = anthropic.Anthropic()
model = "claude-sonnet-4-6"

# Subagent 1: Security Auditor
def run_security_subagent(target: str) -> str:
    response = client.messages.create(
        model=model,
        max_tokens=100,
        system="You are a Cybersecurity Auditor. Focus strictly on authentication & encryption.",
        messages=[{"role": "user", "content": f"Audit security risk for: {target}. 2 points max."}]
    )
    return response.content[0].text

# Subagent 2: Performance Auditor
def run_performance_subagent(target: str) -> str:
    response = client.messages.create(
        model=model,
        max_tokens=100,
        system="You are a Systems Performance Engineer. Focus strictly on latency & scalability.",
        messages=[{"role": "user", "content": f"Audit performance risk for: {target}. 2 points max."}]
    )
    return response.content[0].text

# Orchestrator
def run_orchestrator(system_architecture: str):
    # Spawns subagents with isolated contexts
    sec_report = run_security_subagent(system_architecture)
    perf_report = run_performance_subagent(system_architecture)
    
    # Synthesizes results into final response
    synthesis_prompt = f"""
    Synthesize the following reports into an executive brief:
    
    Security Findings:
    {sec_report}
    
    Performance Findings:
    {perf_report}
    """
    
    final = client.messages.create(
        model=model,
        max_tokens=100,
        system="You are a Lead Solutions Architect producing a brief executive summary.",
        messages=[{"role": "user", "content": synthesis_prompt}]
    )
    
    print("--- Final Synthesized Architecture Brief ---")
    print(final.content[0].text)

run_orchestrator("Serverless PostgreSQL connections with AWS Lambda")


```

---

**Architectural Comparison**

| Feature | Single-Agent System | Multi-Agent Architecture |
| --- | --- | --- |
| **Context Window** | Shared single window (grows with every turn) | Isolated windows per subagent (prevents context degradation) |
| **Prompts & Tools** | Generalist system prompt managing all tools | Focused prompts and tailored tool sets per worker |
| **Execution** | Sequential tool loop | Parallel or staged delegation |
| **Best Used For** | Direct tasks, sequential workflows, low latency | Large-horizon research, complex systems analysis, QA handoffs |
