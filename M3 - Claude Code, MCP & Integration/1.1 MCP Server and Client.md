### MCP Server [transport="stdio"]
```python
from mcp.server.fastmcp import FastMCP

# Initialize FastMCP server
mcp = FastMCP("Demo Server")

# 1. Resource: Provides readable state or contextual data to the client
@mcp.resource("config://app")
def get_config() -> str:
    """Return application configuration string."""
    return "app_version=1.0.0\nenvironment=production"

# 2. Tool: Executable function callable by Claude
@mcp.tool()
def add_numbers(a: int, b: int) -> int:
    """Add two numbers together."""
    return a + b

@mcp.tool()
def multiply_numbers(a: int, b: int) -> int:
    """Multiply two numbers together."""
    return a * b


# 3. Prompt: Reusable prompt template
@mcp.prompt()
def summarize_prompt(text: str) -> str:
    """Generate a prompt template for summarization."""
    return f"Please summarize the following text concisely:\n\n{text}"

if __name__ == "__main__":
    mcp.run(transport="stdio")
```

### MCP Client
```python
import asyncio
import os
from anthropic import Anthropic
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

model = "claude-sonnet-4-6"

async def main():
    server_params = StdioServerParameters(
        command="python",
        args=["mcp-server.py"]
    )

    async with stdio_client(server_params) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()

            # --- 1. Read Resource ---
            resource_data = await session.read_resource("config://app")
            print("Resource Output:")
            print(resource_data.contents[0].text, "\n")

            # --- 2. Retrieve Prompt ---
            prompt_data = await session.get_prompt("summarize_prompt", arguments={"text": "MCP standardizes contextual data transfer."})
            print("Prompt Output:")
            print(prompt_data.messages[0].content.text, "\n")

            # --- 3. Discover and Execute Tools via Claude ---
            tools_response = await session.list_tools()

            print("Available Tools:")
            for tool in tools_response.tools:
                print(f"- {tool.name}: {tool.description}")
            print("\n")

            claude_tools = [
                {
                    "name": tool.name,
                    "description": tool.description,
                    "input_schema": tool.inputSchema
                }
                for tool in tools_response.tools
            ]

            anthropic = Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))
            user_message = "Can you add 15 and 27 together for me?"
            
            response = anthropic.messages.create(
                model=model,
                max_tokens=1000,
                messages=[{"role": "user", "content": user_message}],
                tools=claude_tools
            )

            tool_use = next((block for block in response.content if block.type == "tool_use"), None)
            
            if tool_use:
                print(f"Claude requested tool '{tool_use.name}' with args: {tool_use.input}")
                
                result = await session.call_tool(tool_use.name, tool_use.input)
                print(f"Server Tool Execution Result: {result.content[0].text}")

if __name__ == "__main__":
    asyncio.run(main())
```
